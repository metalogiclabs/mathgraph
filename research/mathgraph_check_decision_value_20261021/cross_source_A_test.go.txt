package toml
import ("bytes";"testing")
func TestMathGraphCrossSourceStringInvariant(t *testing.T) {
 values:=[]string{"plain"," leading","trailing ","has.dot","a=b","has#hash","quote\"value","slash\\value","unicode λ"}
 for _,v:=range values{
  src:=struct { Value string }{Value:v}
  var out bytes.Buffer
  if err:=NewEncoder(&out).Encode(src);err!=nil{t.Fatalf("encode %q: %v",v,err)}
  var decoded struct{Value string}
  if _,err:=Decode(out.String(),&decoded);err!=nil{t.Fatalf("decode %q: %v",v,err)}
  if decoded.Value!=v{t.Fatalf("different value input %q got %q",v,decoded.Value)}
 }
}
